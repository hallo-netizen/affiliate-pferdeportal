294         $token = (string) $token;
295         if (strlen($token) < 6) { return $token; }
296         foreach (array('ern','em','en','er','es','e','n','s') as $suffix) {
297             if ($suffix === 's' && substr($token, -2) === 'ss') { continue; }
298             // "Malene" is not an inflection of "Malen".
299             // Do not turn arbitrary "-ene" tokens into a German infinitive.
300             if ($suffix === 'e' && substr($token, -3) === 'ene') { continue; }
301             if (substr($token, -strlen($suffix)) === $suffix && strlen($token) - strlen($suffix) >= 5) {
302                 return substr($token, 0, -strlen($suffix));
303             }
304         }
