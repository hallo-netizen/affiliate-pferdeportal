2967             $facts[]='Farbe: '.$m[1];
2968         }
2969         if (preg_match('/\b(edelstahl|baumwolle|aluminium|kunststoff|acryl|leder)\b/ui',$title,$m)) {
2970             $facts[]='Material: '.$m[1];
2971         }
2972         // Every other fallback is clearly identified as *topic, price or
2973         // source*, not a claimed special feature. Only use topic if the
2974         // product title independently contains the hobby words.
2975         if (count($facts)<2 && $this->hd_idealo_category_title_relevant($title,$context)) {
2976             $topic=trim((string)($context['primary_name']??''));
2977             if ($topic==='') { $topic=str_replace('-', ' ',sanitize_title((string)($context['primary_slug']??''))); }
2978             if ($topic!=='') { $facts[]='Thema: '.$topic; }
2979         }
2980         if (count($facts)<2 && trim((string)$price)!=='') {
2981             $facts[]='Preis: '.trim((string)$price).' '.strtoupper(substr(sanitize_text_field((string)$currency),0,3));
2982         }
2983         if (count($facts)<2 && in_array(sanitize_key((string)$network),array('idealo','amazon','ebay','otto'),true)) {
2984             $facts[]='Anbieter: '.ucfirst(sanitize_key((string)$network));
2985         }
2986         return array_slice(array_values(array_unique($facts)),0,3);
2987     }

2989     private function hd_idealo_category_title_relevant($title,$context) {
2990         $title=trim(wp_strip_all_tags((string)$title));
2991         $topic=trim((string)($context['primary_name']??''));
2992         if ($topic==='') { $topic=str_replace('-', ' ',sanitize_title((string)($context['primary_slug']??''))); }
2993         if ($title==='' || $topic==='') { return false; }
2994         $need=$this->idealo_match_distinct_tokens($topic,true);
2995         if (!$need) { return false; }
2996         $found=array();
2997         foreach ($this->idealo_match_tokens($title) as $word) {
2998             $found[$this->idealo_match_stem($word)]=true;
2999         }
3000         foreach ($need as $word) {
3001             $stem=$this->idealo_match_stem($word);
3002             if (isset($found[$stem])) { continue; }
3003             // A literal compound such as Airbrushpistole is relevant.
3004             // One-letter Malene/Malen substrings remain rejected.
3005             $compound=false;
3006             foreach (array_keys($found) as $term) {
3007                 if (strlen($stem)>=5 && strpos($term,$stem)===0
3008                     && strlen($term)-strlen($stem)>=4) { $compound=true; break; }
3009             }
3010             if (!$compound) { return false; }
3011         }
3012         return true;
3013     }
