        return array_values(array_unique($classes));
    }

    /**
     * V1.50.465 / Contract V100 – vollständige sichtbare Root-Karte am Astra-Entry-Top.
     * Dadurch hängt die sichtbare Gestaltung nicht mehr von wechselndem Astra-Archivmarkup ab.
     */
    public static function render_journal_category_root_card_v150465() {
        if (is_admin() || !is_category() || !is_main_query()) { return; }
        if (function_exists('in_the_loop') && !in_the_loop()) { return; }
        $term = get_queried_object();
        if (!self::journal_category_image_gap_scope_v150420($term)) { return; }
        $post_id = absint(get_the_ID());
        if ($post_id <= 0 || get_post_type($post_id) !== 'post') { return; }
        $url = get_permalink($post_id);
        if (!$url) { return; }

        $title = (string)get_the_title($post_id);
        $excerpt = trim(wp_strip_all_tags((string)get_the_excerpt($post_id)));
        if ($excerpt === '') {
            $post_obj = get_post($post_id);
            $excerpt = $post_obj ? wp_trim_words(wp_strip_all_tags((string)$post_obj->post_content), 24, ' …') : '';
        }
        $label = trim((string)($term->name ?? ''));

        echo '<a class="pftk-journal-post-image-v150280 pftk-journal-category-owned-v150465" href="' . esc_url($url) . '">';
        if (has_post_thumbnail($post_id)) {
            echo get_the_post_thumbnail($post_id, 'medium_large', ['loading' => 'lazy', 'alt' => $title]);
        } else {
            echo '<span aria-hidden="true">' . self::journal_icon_svg('horse') . '</span>';
        }
        echo '</a><div class="pftk-journal-post-copy-v150280 pftk-journal-category-owned-v150465">';
        if ($label !== '') { echo '<span class="pftk-journal-post-meta-v150280">' . esc_html($label) . '</span>'; }
        echo '<h3><a href="' . esc_url($url) . '">' . esc_html($title) . '</a></h3>';
        if ($excerpt !== '') { echo '<p>' . esc_html($excerpt) . '</p>'; }
        echo '<a class="pftk-journal-button-v150280" href="' . esc_url($url) . '">Weiterlesen</a></div>';
    }
