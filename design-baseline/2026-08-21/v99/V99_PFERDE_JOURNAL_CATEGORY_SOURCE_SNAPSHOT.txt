    /**
     * V1.50.464: genau ein Weiterlesen-CTA in registrierten Journal-Kategoriearchiven.
     * Der offizielle Astra-Hook liegt innerhalb von .entry-content direkt nach dem Auszug.
     */
    public static function render_journal_category_meta_v150464() {
        if (is_admin() || !is_category() || !is_main_query()) { return; }
        if (function_exists('in_the_loop') && !in_the_loop()) { return; }
        $term = get_queried_object();
        if (!self::journal_category_image_gap_scope_v150420($term)) { return; }
        $post_id = absint(get_the_ID());
        if ($post_id <= 0 || get_post_type($post_id) !== 'post') { return; }
        $label = trim((string)($term->name ?? ''));
        if ($label === '') { return; }
        echo '<div class="pftk-journal-category-meta-v150464">' . esc_html($label) . '</div>';
    }

    public static function render_journal_category_read_more_v150464() {
        if (is_admin() || !is_category() || !is_main_query()) { return; }
        $term = get_queried_object();
        if (!self::journal_category_image_gap_scope_v150420($term)) { return; }
        $post_id = absint(get_the_ID());
        if ($post_id <= 0 || get_post_type($post_id) !== 'post') { return; }
        $url = get_permalink($post_id);
        if (!$url) { return; }
        echo '<p class="read-more pftk-journal-category-read-more-v150464"><a href="' . esc_url($url) . '">Weiterlesen</a></p>';
    }
